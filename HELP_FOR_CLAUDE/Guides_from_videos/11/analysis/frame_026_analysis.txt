    caster:Heal(caster:GetSpecialValueFor("heal"), self)

end

modifier_healing_spell = class({})

function modifier_healing_spell:DeclareFunctions()
    return {MODIFIER_PROPERTY_HEALTH_REGEN_CONSTANT}
end

end
