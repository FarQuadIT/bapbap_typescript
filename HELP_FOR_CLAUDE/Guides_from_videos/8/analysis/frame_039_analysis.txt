function ChupakabraMode:InitGameMode()
    GameRules:SetHeroRespawnEnabled( true )
    GameRules:SetUseUniversalShopMode( true )
    GameRules:SetSameHeroSelectionEnabled( true )
    GameRules:SetHeroSelectionTime( 0.0 )
    GameRules:SetPreGameTime( 0.0 )
    GameRules:SetPostGameTime( 60.0 )
    GameRules:SetTreeRegrowTime( 60.0 )
    GameRules:SetUseCustomHeroXPValues ( true ) -- Установка кастомной системы уровней
    GameRules:SetCustomHeroMaxLevel ( #HeroExpTable )
    GameRules:GetGameModeEntity():SetUseCustomHeroLevels ( true )
    GameRules:GetGameModeEntity():SetCustomHeroMaxLevel ( #HeroExpTable )
    GameRules:GetGameModeEntity():SetCustomXPRequiredToReachNextLevel( HeroExpTable )

    GameRules:GetGameModeEntity():SetFogOfWarDisabled( true )
    ListenToGameEvent("entity_killed", Dynamic_Wrap(self, 'ChupakabraKilled'), self)
    ListenToGameEvent("game_rules_state_change", Dynamic_Wrap(self, 'ChupakabraStateChange'), self)
end

function ChupakabraMode:ChupakabraStateChange(data)
    local newState = GameRules:State_Get()
    if newState == DOTA_GAMERULES_STATE_GAME_IN_PROGRESS then
        ChupakabraBroadcast()
    end
end

function ChupakabraMode:ChupakabraKilled(data)
    local killed_unit = EntIndexToHScript(data.entindex_killed)

    if killed_unit:GetUnitName() == "npc_dota_neutral_kobold_boss" then
        local point = Entities:FindByName(nil, "bosses_point"):GetAbsOrigin()
        CreateUnitByName("npc_dota_neutral_ourinour_boss", point, true, nil, nil, DOTA_TEAM_BADGUYS)
    end

    if killed_unit:GetUnitName() == "npc_dota_neutral_ourinour_boss" then
        local point = Entities:FindByName(nil, "bosses_point"):GetAbsOrigin()
        CreateUnitByName("npc_dota_neutral_centaur_outrunner_boss", point, true, nil, nil, DOTA_TEAM_BADGUYS)
    end

    if killed_unit:GetUnitName() == "npc_dota_neutral_centaur_outrunner_boss" then
        local point = Entities:FindByName(nil, "bosses_point"):GetAbsOrigin()
        CreateUnitByName("npc_dota_neutral_fel_beast_boss", point, true, nil, nil, DOTA_TEAM_BADGUYS)
    end

    if killed_unit:GetUnitName() == "npc_dota_neutral_fel_beast_boss" then
        GameRules:SetGameWinner(DOTA_TEAM_GOODGUYS)
    end
end

function GiveGoldToAllPlayers(gold)
    for index = 0, PlayerResource:GetPlayerCount() - 1 do
        if PlayerResource:IsValidPlayer(index) then
            local player = PlayerResource:GetPlayer(index)
            local hero = PlayerResource:GetSelectedHeroEntity(index)
            hero:ModifyGold(gold, false, 0)
            SendOverheadEventMessage( player, OVERHEAD_ALERT_GOLD, hero, gold, nil )
        end
    end
end
