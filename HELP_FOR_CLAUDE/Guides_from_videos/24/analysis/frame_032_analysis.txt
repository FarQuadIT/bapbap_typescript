#### 1. Код

- **Язык программирования**: Lua
- **Видимый код**:
  ```lua
  CustomGameEventManager:RegisterListener("spawn-cobold", Dynamic_Wrap(self, 'OnSpawnCobold'))
  CustomGameEventManager:RegisterListener("add_ability", Dynamic_Wrap(self, 'OnChoiceAbility'))

  function ChupakabraMode:OnSpawnCobold(data)
      local player = PlayerResource:GetPlayer(data.id)
      local hero = player:GetAssignedHero()

      hero:ModifyGold(START_GOLD, true, 1)
  end

  function ChupakabraMode:OnChoiceAbility(data)
      local player = PlayerResource:GetPlayer(data.player)
      local hero = player:GetAssignedHero()

      hero:AddAbility(data.ability)
  end

  function ChupakabraMode:ChupakabraStateChange(data)
      local newState = GameRules:State_Get()
      if newState == DOTA_GAMERULES_STATE_GAME_IN_PROGRESS then
          self:ChupakabraStateInit()
      elseif newState == DOTA_GAMERULES_STATE_CUSTOM_GAME_SETUP then
          ChupakabraMode:InitBossKill()
      end
  end

  function ChupakabraMode:DamageFilter(data)
      return true
  end

  function ChupakabraMode:OrderFilter(data)
      local type = data.order_type
      local playerId = data.issuer_player_id_const

      if type == DOTA_UNIT_ORDER_PICKUP_ITEM then
          local target = EntIndexToHScript(data.entindex_target)

          if target:GetContainedItem():GetCost() > 500 then
              CustomGameEventManager:Send_ServerToPlayer(PlayerResource:GetPlayer(playerId), "ShowError", {message = "Не трогай, а то сожрет"})
              return false
          end
      end

      return true
  end

  function ChupakabraMode:ChupakabraKilled(data)
      local killed_unit = EntIndexToHScript(data.entindex_killed)

      if killed_unit:GetUnitName() == "npc_dota_neutral_kobold_boss" then
          CustomNetTables:SetTableValue("boss_kill", tostring(1), {boss_killed = 0})

          EmitSoundOn("Hero_Abaddon.DeathShield.Cast", killed_unit)
          AddFOWViewer(DOTA_TEAM_GOODGUYS, killed_unit:GetAbsOrigin(), 500, 3, false)
      end

      if killed_unit:GetUnitName() == "npc_dota_neutral_centaur_ourtrunner_boss" then
          local point = Entities:FindByName(nil, "boss_point"):GetAbsOrigin()
          CreateUnitByName("npc_dota_neutral_kobold_boss", point, true, nil, nil, DOTA_TEAM_BADGUYS)
      end

      if killed_unit:GetUnitName() == "npc_dota_neutral_fel_beast_boss" then
          GameRules:SetGameWinner(DOTA_TEAM_GOODGUYS)
      end
  end
  ```

- **Описание кода**:
  - **Регистрация событий**: Используются `CustomGameEventManager:RegisterListener` для регистрации событий, таких как "spawn-cobold" и "add_ability".
  - **Функции**:
    - `OnSpawnCobold`: Добавляет золото герою.
    - `OnChoiceAbility`: Добавляет способность герою.
    - `ChupakabraStateChange`: Инициализирует состояние игры в зависимости от текущего состояния.
    - `DamageFilter`: Возвращает `true`, что позволяет всем видам урона проходить.
    - `OrderFilter`: Проверяет заказы игроков, например, не позволяет поднимать предметы дороже 500 золота.
    - `ChupakabraKilled`: Обрабатывает события убийства различных боссов, включая обновление таблиц и создание новых юнитов.

- **Важные детали**:
  - Используются функции для работы с игроками и героями, такие как `PlayerResource:GetPlayer` и `GetAssignedHero`.
  - Применяются сетевые таблицы через `CustomNetTables:SetTableValue`.
  - Используются звуковые эффекты и эффекты обзора через `EmitSoundOn` и `AddFOWViewer`.
