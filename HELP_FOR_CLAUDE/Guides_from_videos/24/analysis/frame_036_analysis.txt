    local killed_unit = EntIndexToHScript(data.entindex_killed)
    local attacker = EntIndexToHScript(data.entindex_attacker)

    if killed_unit:GetUnitName() == "npc_dota_neutral_kobold_boss" then
        local bossKilled = CustomNetTables:GetTableValue("boss_kill", tostring(attacker:GetPlayerOwnerID()))
        bossKilled = bossKilled + 1
        CustomNetTables:SetTableValue("boss_kill", tostring(attacker:GetPlayerOwnerID()), bossKilled)

        EmitSoundOn("Hero_LegionCommander.Duel.Victory", attacker)
        AddFOWViewer(attacker:GetTeamNumber(), killed_unit:GetAbsOrigin(), 500, 3, false)

        if killed_unit:IsAlive() then
            killed_unit:RemoveSelf()
        end
    end
end
