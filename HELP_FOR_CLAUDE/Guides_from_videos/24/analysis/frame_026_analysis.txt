### Видимый код:

```lua
local hero = Player:GetAssignedHero()
hero:ModifyGold(START_GOLD, true, 13)

function ChupakabraMode:OnNPCSpawned(data)
    local player = PlayerResource:GetPlayer(data.player)
    local hero = Player:GetAssignedHero()
    hero:AddAbility(data.ability)
end

function ChupakabraMode:ChupakabraStateChange(data)
    local newState = GameRules:State_Get()
    if newState == DOTA_GAMERULES_STATE_CUSTOM_GAME_SETUP then
        ChupakabraMode:InitBossKill()
    end
end

function ChupakabraMode:DamageFilter(data)
    return true
end

function ChupakabraMode:OrderFilter(data)
    local type = data.order_type
    local playerId = data.issuer_player_id_const

    if type == DOTA_UNIT_ORDER_PICKUP_ITEM then
        local target = EntIndexToHScript(data.entindex_target)

        if target:GetContainedItem():GetCost() > 500 then
            CustomGameEventManager:Send_ServerToPlayer(PlayerResource:GetPlayer(playerId), "ShowError", {message = "Не трогай, а то сожрет"})
            return false
        end
    end

    return true
end

function ChupakabraMode:ChupakabraKilled(data)
    local killed_unit = EntIndexToHScript(data.entindex_killed)

    if killed_unit:GetUnitName() == "npc_dota_neutral_kobold_boss" then
        EmitSoundOn("Hero_Abaddon.AphoticShield.Cast", killed_unit)
        AddItemAndNotifyForAll()
    end

    if killed_unit:GetUnitName() == "npc_dota_neutral_centaur_outrunner_boss" then
        local point = Entities:FindByName(nil, "bosses_point"):GetAbsOrigin()
        CreateUnitByName("npc_dota_neutral_beast_boss", point, true, nil, nil, DOTA_TEAM_BADGUYS)
        AddRandomAbilityForUnit()
    end

    if killed_unit:GetUnitName() == "npc_dota_neutral_fel_beast_boss" then
        GameRules:SetCustomGameEndTime(DOTA_TEAM_GOODGUYS)
    end
end

function ChupakabraMode:ChupakabraModeStart()
    local hero = PlayerResource:GetPlayer()
    local player = PlayerResource:GetPlayer()
    local hero = Player:GetAssignedHero()
    hero:AddAbility("ability_custom")
end
```
