```javascript
function createAbilityPanel(event) {
    const arrayAbilities = Object.values(event.abilities)
    const rootPanel = $.CreatePanel("Panel", $.GetContextPanel(), "black-cube")
    rootPanel.BLoadLayoutSnippet("ability")

    rootPanel.AddClass("IsActive")

    const panelBody = rootPanel.FindChildTraverse("black-cube__body")

    arrayAbilities.map((element) => {
        const ability = $.CreatePanel("DOTAAbilityImage", panelBody, "", {
            class: "black-cube__ability",
            abilityname: element,
            showtooltip: "true"
        })

        ability.SetPanelEvent("onactivate", () => {
            GameEvents.SendCustomGameEventToServer("add_ability", {
                player: Players.GetLocalPlayer(),
                ability: element
            })
            rootPanel.RemoveClass("IsActive")
            rootPanel.DeleteAsync(0.4)
        })
    })
}

GameEvents.Subscribe("show_ability_panel", createAbilityPanel)

const dotaHud = (() => {
    let panel = $.GetContextPanel();
    while (panel) {
        if (panel.id === "DotaHud")
            return panel;
        panel = panel.GetParent();
    }
    return panel;
})();

const talent = dotaHud.FindChildTraverse("StatBranch");
const levels = dotaHud.FindChildTraverse("level_stats_frame");
talent.style.opacity = "0";
talent.enabled = false;
levels.style.opacity = "0";
```
