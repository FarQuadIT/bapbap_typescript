end

modifier_strong_poison = class({})

function modifier_strong_poison:IsHidden()
    return false
end

function modifier_strong_poison:DeclareFunctions()

        MODIFIER_PROPERTY_MOVESPEED_BONUS_CONSTANT
    }
end

function modifier_strong_poison:GetModifierMoveSpeedBonus_Constant()

end

function modifier_strong_poison:OnCreated()
    self:StartIntervalThink(0.5)
end

function modifier_strong_poison:OnIntervalThink()

        victim = self:GetParent(),
        attacker = self:GetCaster(),
        damage = 50,
        damage_type = DAMAGE_TYPE_MAGICAL,
        ability = self:GetAbility()
    })
end

function modifier_strong_poison:GetEffectName()
