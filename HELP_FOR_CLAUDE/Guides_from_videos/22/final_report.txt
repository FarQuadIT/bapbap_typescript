        bProvidesVision = true,
        iVisionTeamNumber = caster:GetTeamNumber(),
        iVisionRadius = 500,
        ExtraData = {
            counter = 1,
            name = "asdas",
            boolean = true,
        }
    })
end

function wave_terorist:OnProjectileHit_ExtraData(target, _, extraData)
    local caster = self:GetCaster()

    if not target then
        caster:Heal(9999, self)
    end

    ApplyDamage({
        victim = target,
        attacker = caster,
        ability = self,
        damage = 100,
        damage_type = DAMAGE_TYPE_MAGICAL,
    })
end
