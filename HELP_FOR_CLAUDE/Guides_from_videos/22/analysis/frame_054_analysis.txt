```lua
wave_terorist = class({})

function wave_terorist:OnSpellStart()
    local caster = self:GetCaster()
    local point = self:GetCursorPosition()
    local casterPoint = caster:GetOrigin()

    local derction = point - casterPoint
    local speed = 1000
    derction = derction:Normalized()

    ProjectileManager:CreateLinearProjectile({
        EffectName = "particles/units/heroes/hero_vengeful/vengeful_wave_of_terror.vpcf",
        Ability = self,
        Source = caster,
        vSpawnOrigin = casterPoint,
        fStartRadius = 200,
        fEndRadius = 200,
        vVelocity = derction * speed,
        fDistance = 1000,
        UnitBehavior = PROJECTILES_NOTHING,
        UnitTest = function(self, unit) return unit:GetTeamNumber() ~= caster:GetTeamNumber() end,
        OnUnitHit = function(self, unit)
            ApplyDamage({
                victim = unit,
                attacker = caster,
                damage = 100,
                damage_type = DAMAGE_TYPE_MAGICAL,
            })
            unit:AddNewModifier(caster, self, "modifier_stunned", {duration = 1})
        end,
    })
end
```
