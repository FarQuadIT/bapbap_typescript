```lua
wave_terorist = class({})

function wave_terorist:OnSpellStart()
    local caster = self:GetCaster()
    local point = self:GetCursorPosition()
    local casterPoint = caster:GetOrigin()

    local direction = point - casterPoint
    local speed = 1600

    direction = direction:Normalized()

    ProjectileManager:CreateLinearProjectile({
        EffectName = "particles/units/heroes/hero_vengeful/vengeful_wave_of_terror.vpcf",
        Ability = self,
        Source = caster,
        vSpawnOrigin = casterPoint,
        fStartRadius = 200,
        fEndRadius = 200,
        vVelocity = direction * speed,
        fDistance = 2000,
        UnitTargetTeam = DOTA_UNIT_TARGET_TEAM_ENEMY,
        UnitTargetFlags = DOTA_UNIT_TARGET_FLAG_NONE,
        UnitTargetType = DOTA_UNIT_TARGET_HERO + DOTA_UNIT_TARGET_CREEP,
        bProvidesVision = true,
        iVisionTeamNumber = caster:GetTeamNumber(),
        iVisionRadius = 500,
    })
end

function wave_terorist:OnProjectileHit(target)
    if not target then return end
    local caster = self:GetCaster()

    ApplyDamage({
        victim = target,
        attacker = caster,
        ability = self,
        damage = 100,
        damage_type = DAMAGE_TYPE_MAGICAL,
    })

    target:AddNewModifier(caster, self, "modifier_stunned", {duration = 1})
end
```
