```lua
wave_terorist = class({})

function wave_terorist:OnSpellStart()
    local caster = self:GetCaster()
    local point = self:GetCursorPosition()
    local casterPoint = caster:GetOrigin()

    local derction = point - casterPoint
    local speed = 600
    derction = derction:Normalized()

    ProjectileManager:CreateLinearProjectile({
        Ability = self,
        EffectName = "particles/units/heroes/hero_vengeful/vengeful_wave_of_terror.vpcf",
        vSpawnOrigin = casterPoint,
        fStartRadius = 200,
        fEndRadius = 200,
        vVelocity = derction * speed,
        fDistance = 1200,
        Source = caster,
        iUnitTargetTeam = DOTA_UNIT_TARGET_TEAM_ENEMY,
        iUnitTargetFlags = DOTA_UNIT_TARGET_FLAG_NONE,
        iUnitTargetType = DOTA_UNIT_TARGET_HERO + DOTA_UNIT_TARGET_CREEP,
        bProvidesVision = true,
        iVisionTeamNumber = caster:GetTeamNumber(),
        iVisionRadius = 500
    })
end

function wave_terorist:OnProjectileHit(target)
    local caster = self:GetCaster()

    if not target then
        caster:Heal(9999, self)
        return
    end

    ApplyDamage({
        victim = target,
        attacker = caster,
        ability = self,
        damage = 100,
        damage_type = DAMAGE_TYPE_MAGICAL,
    })

    target:AddNewModifier(caster, self, "modifier_stunned", {duration = 1})
end
```

- **Функция `OnProjectileHit`:** Выполняется при попадании снаряда. Если цель не найдена, исцеляет кастера на 9999 единиц. Если цель есть, наносит ей магический урон и оглушает на 1 секунду.

### Важные детали:
