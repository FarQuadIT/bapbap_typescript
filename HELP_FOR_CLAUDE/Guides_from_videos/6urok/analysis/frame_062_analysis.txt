function chupakabramode:InitGameMode()
    GameRules:SetStartingGold(1500)
    GameRules:SetHeroSelectionTime(0.0)
    GameRules:SetPreGameTime(10.0)
    GameRules:SetPostGameTime(10.0)
    GameRules:SetTreeRegrowTime(60.0)
    GameRules:SetUseCustomHeroXPValues(true) -- Установка кастомной системы уровней
    GameRules:SetHeroRespawnEnabled(false)
    GameRules:SetCustomAttributeDerivedStatValue(DOTA_ATTRIBUTE_AGILITY_ARMOR, 10.0)
    GameRules:SetCustomAttributeDerivedStatValue(DOTA_ATTRIBUTE_AGILITY_ATTACK_SPEED, 50.0)

    ListenToGameEvent("entity_killed", Dynamic_Wrap(self, 'ChupakabraKilled'), self)
    GameRules:GetGameModeEntity():SetThink("GlobalThink", self, "GlobalThink", 2)
end

function chupakabramode:EvaluateGameState()
    if GameRules:State_Get() == DOTA_GAMERULES_STATE_GAME_IN_PROGRESS then
        -- Check if template addon script is running.
        if GameRules:State_Get() >= DOTA_GAMERULES_STATE_POST_GAME then
            return nil
        end
        return 1
    end
end

local kobold_count = 0

function chupakabramode:ChupakabraKilled(data)
    local killed_unit = EntIndexToHScript(data.entindex_killed)

    if killed_unit:GetUnitName() == "npc_dota_neutral_kobold" then
        kobold_count = kobold_count + 1
        CreateItemOnPositionSync(killed_unit:GetAbsOrigin() + RandomVector(RandomFloat(50, 200)), nil)
    end

    if kobold_count >= 10 then
        GameRules:SetGameWinner(DOTA_TEAM_GOODGUYS)
    end

    if killed_unit:IsRealHero() then
        GameRules:SetGameWinner(DOTA_TEAM_BADGUYS)
    end
end
