function InvasionMode:InvasionGameModeStateChange(data)
    local newState = GameRules:State_Get()
    if newState == DOTA_GAMERULES_STATE_GAME_IN_PROGRESS then
        print("Game In Progress")
        InvadeInvasionGameState()
    elseif newState == DOTA_GAMERULES_STATE_POST_GAME then
        print("Post Game")
        return
    end
end

        kobold_count = kobold_count + 1
        print("Kobold killed: ", kobold_count)
        CreateDropItem("item_lots_of_gold", killed_unit:GetAbsOrigin())

    if killed_unit:GetTeam() == DOTA_TEAM_GOODGUYS then
        GameRules:SetCustomGameTeamMaxPlayers(DOTA_TEAM_GOODGUYS, 0)
