local kobold_count = 0

    local killed_unit = EntIndexToHScript(data.entindex_killed)
    if killed_unit:GetUnitName() == "npc_dota_neutral_kobold" then
        kobold_count = kobold_count + 1
        CreateDrop("item_bag_of_gold", killed_unit:GetAbsOrigin() + RandomVector(RandomFloat(50, 200)))
    end
    if kobold_count >= 10 then
        GameRules:SetGameWinner(DOTA_TEAM_GOODGUYS)
    end
    if killed_unit:IsRealHero() then
        GameRules:SetGameWinner(DOTA_TEAM_BADGUYS)
    end

function GiveGoldPlayers(gold)
    for index = 0, PlayerResource:GetPlayerCount() do
        if PlayerResource:IsValidPlayer(index) then
            local player = PlayerResource:GetPlayer(index)
            local hero = player:GetAssignedHero()
            hero:ModifyGold(gold, false, 0)
            SendOverheadEventMessage(player, OVERHEAD_ALERT_GOLD, hero, gold, nil)
        end
    end
end

function CreateDrop(itemname, pos)
    local newItem = CreateItem(itemname, nil, nil)
    local drop = CreateItemOnPositionSync(pos, newItem)
    newItem:LaunchLoot(false, 300, 0.75, pos + RandomVector(RandomFloat(50, 350)))
end
