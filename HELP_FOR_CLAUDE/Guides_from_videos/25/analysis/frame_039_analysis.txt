  ```lua
  if require("tables/item_drop") then
      if ItemDropManager == nil then
          ItemDropManager = class({})
      end

      function ItemDropManager:Init()
          ListenToGameEvent("entity_killed", Dynamic_Wrap(self, 'EntityKilled'), self)
      end

      function ItemDropManager:EntityKilled(event)
          local killedUnit = EntIndexToHScript(event.entindex_killed)
          local items = ITEM_DROP[killedUnit:GetUnitName()]

          if not items then return end

          local randomNumber = RandomFloat(0, 99)
          local cumulativeProbability = 0
          local dropItem = nil

          for _, item in pairs(items) do
              cumulativeProbability = cumulativeProbability + item.chance
              if randomNumber < cumulativeProbability then
                  dropItem = item
                  break
              end
          end
      end
  end
  ```

- **Описание кода**:
  - Код реализует систему выпадения предметов в игре.
  - Используется класс `ItemDropManager` для управления событиями выпадения предметов.
  - Функция `Init` подписывается на событие `entity_killed`, чтобы вызывать `EntityKilled` при убийстве сущности.
  - В `EntityKilled` определяется, какие предметы могут выпасть из убитой сущности, и случайным образом выбирается один из них на основе вероятности.

- **Важные детали**:
  - Используется функция `require` для подключения модуля `tables/item_drop`.
  - Класс `ItemDropManager` создается только если он еще не определен.
  - Вероятность выпадения предмета определяется через `RandomFloat` и суммируется в `cumulativeProbability`.
