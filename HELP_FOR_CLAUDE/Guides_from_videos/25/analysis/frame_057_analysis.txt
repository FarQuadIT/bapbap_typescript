### Код:
```lua
require("tables/item_drop")

if ItemDropManager == nil then
    ItemDropManager = class({})
end

function ItemDropManager:init()
    ListenToGameEvent("entity_killed", Dynamic_Wrap(self, 'EntityKilled'), self)
end

function ItemDropManager:EntityKilled(event)
    local killedUnit = EntIndexToHScript(event.entindex_killed)
    local items = ITEM_DROP[killedUnit:GetUnitName()]

    if not items then return end

    local randomNumber = RandomFloat(0, 0.99)
    local cumulativeProbability = 0
    local droptItem

    for _, item in ipairs(items) do
        cumulativeProbability = cumulativeProbability + item.chance
        if randomNumber < cumulativeProbability then
            dropItem = item
            break
        end
    end

    if not dropItem then return end

    local spawnPoint = killedUnit:GetAbsOrigin()
    local newItem = CreateItem(dropItem.item, nil, nil)
    CreateItemOnPositionForLaunch(spawnPoint, newItem)
    newItem:LaunchLootInitialHeight(false, 0, 150, 0.5, spawnPoint + RandomVector(150))
end
```
