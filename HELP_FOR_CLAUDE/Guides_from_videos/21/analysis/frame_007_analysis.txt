    ListenToGameEvent('entity_killed', Dynamic_Wrap(self, 'ChupakabraKilled'), self)

function ChupakabraMode:OnSpawnCobalt(data)
    local player = PlayerResource:GetPlayer(data.PlayerID)
    local hero = player:GetAssignedHero()

    hero:ModifyGold(START_GOLD, true, 0)
end

function ChupakabraMode:OnChoiceAbility(data)
    local player = PlayerResource:GetPlayer(data.PlayerID)
    local hero = player:GetAssignedHero()

    hero:AddAbility(data.ability)
end
