end

        self:ChupakabraModeStart()
    end
end

    return true
end

        local target = EntIndexToHScript(data.entindex_target)

        if target:GetContainedItem():GetCost() >= 500 then
            CustomGameEventManager:Send_ServerToPlayer(EntityPlayer, "ShowError", {message = "Не трогай, а то сожрет"})
