  ```lua
  function Spawn()
      if not thisEntity then return nil end

      thisEntity:SetContextThink("KoboldThink", KoboldThink, 1)
  end

  function KoboldThink()
      if GameRules:State_Get() == DOTA_GAMERULES_STATE_POST_GAME or not thisEntity:IsAlive() then
          return nil
      end

      if thisEntity:GetHealth() < thisEntity:GetMaxHealth() * 0.5 then
          local item = PlayerResource:GetPlayer(0):GetAssignedHero():FindItemInInventory("item_radiance")
          if item then
              ExecuteOrderFromTable({
                  UnitIndex = PlayerResource:GetPlayer(0):GetAssignedHero():entindex(),
                  OrderType = DOTA_UNIT_ORDER_DROP_ITEM,
                  AbilityIndex = item:entindex(),
                  Position = PlayerResource:GetPlayer(0):GetAssignedHero():GetAbsOrigin()
              })
          end
      end

      return 1
  end
  ```
