function KoboldThink()
    if GameRules:State_Get() == DOTA_GAMERULES_STATE_POST_GAME or not thisEntity:IsAlive() then
        return nil
    end

    if thisEntity:GetHealth() < thisEntity:GetMaxHealth() * 0.5 then
        local ability = thisEntity:FindAbilityByName("alchemist_chemical_rage")

        if ability:IsFullyCastable() then
            ExecuteOrderFromTable({
                UnitIndex = thisEntity:entindex(),
                OrderType = DOTA_UNIT_ORDER_CAST_NO_TARGET,
                AbilityIndex = ability:entindex(),
            })
        end
    end

    print("dsds")
    return 1
end
