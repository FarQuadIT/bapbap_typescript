### Видимый код:
```lua
modifier_gold_hole_thinker = class({})

function modifier_gold_hole_thinker:IsHidden()
    return true
end

function modifier_gold_hole_thinker:OnCreated()
    local radius = self:GetAbility():GetSpecialValueFor("radius")

    self.effect_cast = ParticleManager:CreateParticle("particles/econ/items/riki/riki_head_ti8_gold/riki_smokebomb_ti8_gold.vpcf", PATTACH_WORLDORIGIN, self:GetParent())
    ParticleManager:SetParticleControl(self.effect_cast, 0, self:GetParent():GetOrigin())
    ParticleManager:SetParticleControl(self.effect_cast, 1, Vector( radius, 0, 0 ) )

    self:StartIntervalThink(1)
end

function modifier_gold_hole_thinker:OnDestroy()
    ParticleManager:DestroyParticle(self.effect_cast, false)
end

function modifier_gold_hole_thinker:OnIntervalThink()
    local caster = self:GetCaster()
    local parent = self:GetParent()
    local radius = self:GetAbility():GetSpecialValueFor("radius")
    local gold_per_unit = self:GetAbility():GetSpecialValueFor("gold_per_unit")

    if caster:HasScepter() then
        gold_per_unit = 1000
    end

    local enemies = FindUnitsInRadius(caster:GetTeamNumber(), parent:GetAbsOrigin(), nil, radius, DOTA_UNIT_TARGET_TEAM_ENEMY, DOTA_UNIT_TARGET_HERO, 0, 0, false)
    local gold = gold_per_unit * #enemies

    caster:ModifyGold(gold, true, 13)
    EmitSoundOnClient("General.Coins", caster)
    SendOverheadEventMessage(caster:GetPlayerOwner(), OVERHEAD_ALERT_GOLD, caster, gold, caster)
end
```

- **Класс `modifier_gold_hole_thinker`:** Определяет модификатор, который управляет созданием и уничтожением частицы, а также начислением золота.
- **Функция `IsHidden`:** Возвращает `true`, скрывая модификатор.
- **Функция `OnCreated`:** Создает эффект частицы с заданным радиусом и запускает интервал для периодического выполнения.
- **Функция `OnDestroy`:** Уничтожает созданную частицу.
- **Функция `OnIntervalThink`:** Выполняется периодически, проверяет наличие скипетра у кастера и начисляет золото за каждого врага в радиусе.

- **Функция `FindUnitsInRadius`:** Находит врагов в заданном радиусе.
- **Проверка `HasScepter`:** Изменяет количество золота за врага, если у кастера есть скипетр.
